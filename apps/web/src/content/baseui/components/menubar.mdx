## Anatomy

#### Components

```tsx
<MenubarRoot>
  <MenuRoot>
    <MenuTrigger />

    <MenuContent>
      <MenuArrow />
      <MenuItem />
      <MenuShortcut />
      <MenuSeparator />

      <MenuSubmenu>
        <MenuSubmenuTrigger />
        <MenuContent>
          <MenuItem />
        </MenuContent>
      </MenuSubmenu>

      <MenuGroup>
        <MenuGroupLabel />
      </MenuGroup>

      <MenuRadioGroup>
        <MenuRadioItem />
      </MenuRadioGroup>

      <MenuCheckboxItem />
    </MenuContent>
  </MenuRoot>
</MenubarRoot>
```

```tsx
<Menubar>

            <Menu.SubmenuTrigger />
          </Menu.SubmenuRoot>

          <Menu.Group>
            <Menu.GroupLabel />
          </Menu.Group>

          <Menu.RadioGroup>
            <Menu.RadioItem>
              <Menu.RadioItemIndicator />

            <Menu.CheckboxItemIndicator />
          </Menu.CheckboxItem>

          <Menu.Viewport />
        </Menu.Popup>
      </Menu.Positioner>
    </Menu.Portal>
  </Menu.Root>
</Menubar>
```
