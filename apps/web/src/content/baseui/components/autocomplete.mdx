## Anatomy

#### Composite

```tsx
<Autocomplete.Root items={items}>
  <Autocomplete.InputGroup inputProps={{ placeholder: "Search..." }} />

  <Autocomplete.Content>
    <Autocomplete.Empty>No results found.</Autocomplete.Empty>
    <Autocomplete.List>
      {(item) => (
        <Autocomplete.Item key={item.id} value={item}>
          {item.label}
        </Autocomplete.Item>
      )}
    </Autocomplete.List>
  </Autocomplete.Content>
</Autocomplete.Root>
```

#### Primitives (Base UI)

```tsx
<Autocomplete.Root>
  <Autocomplete.InputGroup>
    <Autocomplete.Input />
    <Autocomplete.Trigger />
    <Autocomplete.Icon />
    <Autocomplete.Clear />
    <Autocomplete.Value />
  </Autocomplete.InputGroup>

  <Autocomplete.Portal>
    <Autocomplete.Backdrop />
    <Autocomplete.Positioner>
      <Autocomplete.Popup>
        <Autocomplete.Arrow />

        <Autocomplete.Status />
        <Autocomplete.Empty />

        <Autocomplete.List>
          <Autocomplete.Row>
            <Autocomplete.Item />
          </Autocomplete.Row>

          <Autocomplete.Separator />

          <Autocomplete.Group>
            <Autocomplete.GroupLabel />
          </Autocomplete.Group>

          <Autocomplete.Collection />
        </Autocomplete.List>
      </Autocomplete.Popup>
    </Autocomplete.Positioner>
  </Autocomplete.Portal>
</Autocomplete.Root>
```
