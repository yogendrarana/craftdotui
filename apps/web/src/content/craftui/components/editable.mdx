## Anatomy

```tsx
<EditableRoot defaultValue="Click to edit">
  <EditableLabel />
  <EditableArea>
    <EditablePreview />
    <EditableInput />
  </EditableArea>
  <EditableControl>
    <EditableEditTrigger />
    <EditableSubmitTrigger />
    <EditableCancelTrigger />
  </EditableControl>
</EditableRoot>
```

```tsx
<Editable.Root defaultValue="Click to edit">
  <Editable.Label />
  <Editable.Area>
    <Editable.Preview />
    <Editable.Input />
  </Editable.Area>
  <Editable.Control>
    <Editable.EditTrigger />
    <Editable.SubmitTrigger />
    <Editable.CancelTrigger />
  </Editable.Control>
</Editable.Root>
```
