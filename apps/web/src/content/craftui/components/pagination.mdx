## Anatomy

#### Usage

```tsx
<PaginationRoot>
  <PaginationContent>
    <PaginationItem>
      <PaginationFirst href="#" />
    </PaginationItem>
    <PaginationItem>
      <PaginationPrevious href="#" />
    </PaginationItem>
    <PaginationItem>
      <PaginationLink href="#" isActive>
        1
      </PaginationLink>
    </PaginationItem>
    <PaginationItem>
      <PaginationEllipsis />
    </PaginationItem>
    <PaginationItem>
      <PaginationNext href="#" />
    </PaginationItem>
    <PaginationItem>
      <PaginationLast href="#" />
    </PaginationItem>
  </PaginationContent>
</PaginationRoot>
```

```tsx
<Pagination.Root>
  <Pagination.Content>
    <Pagination.Item>
      <Pagination.First href="#" />
    </Pagination.Item>
    <Pagination.Item>
      <Pagination.Previous href="#" />
    </Pagination.Item>
    <Pagination.Item>
      <Pagination.Link href="#" isActive>
        1
      </Pagination.Link>
    </Pagination.Item>
    <Pagination.Item>

    </Pagination.Item>
    <Pagination.Item>
      <Pagination.Last href="#" />
    </Pagination.Item>
  </Pagination.Content>
</Pagination.Root>
```
