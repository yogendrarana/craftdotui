## Anatomy

#### Usage

```tsx
<StepperRoot defaultValue={1}>
  <StepperList>
    <StepperItem step={1}>
      <StepperTrigger>
        <StepperIndicator />
        <StepperTitle />
        <StepperDescription />
      </StepperTrigger>
      <StepperSeparator />
    </StepperItem>
  </StepperList>
  <StepperContent step={1} />
  <StepperPrev />
  <StepperNext />
</StepperRoot>
```

```tsx
<Stepper.Root defaultValue={1}>
  <Stepper.List>
    <Stepper.Item step={1}>
      <Stepper.Trigger>
        <Stepper.Indicator />
        <Stepper.Title />
        <Stepper.Description />
      </Stepper.Trigger>
      <Stepper.Separator />
    </Stepper.Item>
  </Stepper.List>
  <Stepper.Content step={1} />
  <Stepper.Prev />
  <Stepper.Next />
</Stepper.Root>
```
